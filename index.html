function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.tanggal || new Date().toISOString().split('T')[0], data.nis || '-', data.nama || '-', data.mapel || 'Bahasa Indonesia', data.tugas || 'LKPD Unsur Intrinsik & Majas', data.nilai || 0 ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
